class Row(val title: String, val sub: String, val color: Int, val tag: Any? = null)

class RowAdapter(private val ctx: Context) : BaseAdapter() {
    var rows: List<Row> = emptyList()
    override fun getCount(): Int = rows.size
    override fun getItem(position: Int): Any = rows[position]
    override fun getItemId(position: Int): Long = position.toLong()
    override fun getView(position: Int, convertView: View?, parent: ViewGroup?): View {
        val lay: LinearLayout
        if (convertView is LinearLayout) lay = convertView
        else {
            lay = LinearLayout(ctx)
            lay.orientation = LinearLayout.VERTICAL
            val pad = (12 * ctx.resources.displayMetrics.density).toInt()
            lay.setPadding(pad, pad - 2, pad, pad - 2)
            val t = TextView(ctx).apply { textSize = 14f; setTypeface(null, Typeface.BOLD) }
            val s = TextView(ctx).apply { textSize = 12f; setTextColor(Color.parseColor("#8B949E")) }
            lay.addView(t); lay.addView(s)
        }
        val r = rows[position]
        (lay.getChildAt(0) as TextView).apply { text = r.title; setTextColor(r.color) }
        (lay.getChildAt(1) as TextView).apply {
            text = r.sub
            visibility = if (r.sub.isEmpty()) View.GONE else View.VISIBLE
        }
        return lay
    }
}

    private fun loadUserApps() {
        Thread {
            val res = ArrayList<Pair<String, String>>()
            try {
                for (ai in packageManager.getInstalledApplications(0)) {
                    val sys = (ai.flags and ApplicationInfo.FLAG_SYSTEM) != 0 &&
                        (ai.flags and ApplicationInfo.FLAG_UPDATED_SYSTEM_APP) == 0
                    if (sys || ai.packageName == packageName) continue
                    res.add(Pair(ai.packageName, ai.loadLabel(packageManager).toString()))
                }
            } catch (e: Exception) {}
            res.sortBy { it.second.lowercase() }
            runOnUiThread {
                userApps = res
                val labels = ArrayList<String>()
                labels.add("Todas las apps")
                for (a in res) labels.add(a.second)
                val ad = ArrayAdapter(this, android.R.layout.simple_spinner_item, labels)
                ad.setDropDownViewResource(android.R.layout.simple_spinner_dropdown_item)
                spinner.adapter = ad
            }
        }.start()
    }

    private fun toggle() {
        if (Store.running) {
            startService(Intent(this, NetLensVpnService::class.java).setAction(NetLensVpnService.ACTION_STOP))
            return
        }
        if (Build.VERSION.SDK_INT >= 33 &&
            ContextCompat.checkSelfPermission(this, Manifest.permission.POST_NOTIFICATIONS) != PackageManager.PERMISSION_GRANTED
        ) notifLauncher.launch(Manifest.permission.POST_NOTIFICATIONS)
        val p = VpnService.prepare(this)
        if (p != null) vpnLauncher.launch(p) else launchService()
    }

    private fun launchService() {
        val pos = spinner.selectedItemPosition
        val target: String? = if (pos > 0 && pos - 1 < userApps.size) userApps[pos - 1].first else null
        if (Store.mitmEnabled && target == null) {
            Toast.makeText(this, "MITM requiere una app objetivo", Toast.LENGTH_LONG).show(); return
        }
        Store.clear()
        val i = Intent(this, NetLensVpnService::class.java)
        i.action = NetLensVpnService.ACTION_START
        if (target != null) i.putExtra(NetLensVpnService.EXTRA_PKG, target)
        ContextCompat.startForegroundService(this, i)
    }

    private fun selectTab(i: Int) {
        tab = i
        for (k in tabButtons.indices) {
            tabButtons[k].setTextColor(if (k == i) cBg else cText)
            tabButtons[k].setBackgroundColor(if (k == i) cAccent else Color.parseColor("#21262D"))
        }
        if (i == 2) loadPerms()
        refresh()
    }

    private fun loadPerms() {
        if (permLoading) return
        permLoading = true
        Thread {
            permRows = Perms.scan(packageManager, packageName)
            permLoading = false
            runOnUiThread { refresh() }
        }.start()
    }

    private fun fmt(n: Long): String = Insights.fmt(n)

    private fun riskAlerts(): List<Row> {
        val rows = ArrayList<Row>()
        val permsByPkg = permRows.associateBy { it.pkg }
        for (s in Store.apps.values) {
            val p = permsByPkg[s.pkg] ?: continue
            if (s.trackers.isEmpty()) continue
            val sens = p.perms.filter { it.second }.map { it.first }
            val risky = sens.any { it.startsWith("Ubicación") || it == "Micrófono" || it == "Cámara" || it.contains("contactos") }
            if (!risky) continue
            val sub = "Permisos: " + sens.joinToString(", ") + "\nContacta con: " + s.trackers.take(3).joinToString(", ")
            rows.add(Row("⚠ " + s.label, sub, cBad, s))
        }
        return rows
    }

    private fun refresh() {
        val running = Store.running
        btnStart.text = if (running) "■ Detener" else "▶ Iniciar captura"
        spinner.isEnabled = !running
        val flows = Store.snapshot()
        status.text = if (running)
            "● Capturando · " + flows.size + " conexiones · " + Store.apps.size + " apps" + (if (Store.mitmEnabled) " · MITM ON" else "")
        else "○ Detenido · elige una app (o todas) y pulsa Iniciar"
        status.setTextColor(if (running) cOk else cMuted)
        if (paused && tab == 0) return

        val rows = ArrayList<Row>()
        when (tab) {
            0 -> {
                rows.addAll(riskAlerts())
                val now = System.currentTimeMillis()
                var shown = 0
                for (f in flows.asReversed()) {
                    if (hideDns && f.proto == "DNS") continue
                    if (shown >= 300) break
                    shown++
                    val notes = synchronized(f.notes) { ArrayList(f.notes) }
                    val color = when {
                        notes.any { it.startsWith("🔓") } -> cPurple
                        notes.any { it.startsWith("⚠") } -> cBad
                        notes.any { it.startsWith("🎯") } -> cWarn
                        notes.any { it.startsWith("⛔") || it.startsWith("✖") } -> cMuted
                        else -> cText
                    }
                    val ago = (now - f.start) / 1000
                    val line1 = f.proto + " · ↑" + fmt(f.up.get()) + " ↓" + fmt(f.down.get()) + " · hace " + ago + " s" + (if (f.closed) " · cerrada" else "")
                    val line2 = if (notes.isEmpty()) "" else "\n" + notes.take(3).joinToString("\n")
                    rows.add(Row(f.stat.label + " → " + f.name() + ":" + f.dstPort, line1 + line2, color, f))
                }
                if (rows.isEmpty()) rows.add(Row("Sin tráfico todavía", "Pulsa «Iniciar captura», concede el permiso de VPN y abre otra app.", cMuted))
            }
            1 -> {
                val stats = Store.apps.values.sortedByDescending { it.up.get() + it.down.get() }
                for (s in stats) {
                    val bad = s.clearFlows.get() > 0
                    val tr = s.trackers.size
                    val dec = s.mitmCount.get() > 0
                    val color = if (bad) cBad else if (dec) cPurple else if (tr > 0) cWarn else cText
                    var sub = "↑" + fmt(s.up.get()) + " ↓" + fmt(s.down.get()) + " · " + s.hosts.size + " dominios · " + s.flows.get() + " conexiones"
                    if (tr > 0) sub += "\n🎯 " + tr + " destinos de anuncios/analítica/fallos"
                    if (bad) sub += "\n⚠ " + s.clearFlows.get() + " conexiones HTTP sin cifrar"
                    if (dec) sub += "\n🔓 " + s.mitmCount.get() + " conexiones descifradas (MITM)"
                    rows.add(Row(s.label, sub, color, s))
                }
                if (rows.isEmpty()) rows.add(Row("Aún no hay datos", "Inicia la captura y usa alguna app.", cMuted))
            }
            2 -> {
                rows.add(Row("Esta app (NetLens)", "VPN: la concede el sistema. Notificaciones: toca aquí. CA local: expórtala desde el botón «Exportar CA» e instálala en Ajustes → Seguridad.", cAccent, "self"))
                if (permLoading && permRows.isEmpty()) rows.add(Row("Analizando apps instaladas…", "", cMuted))
                for (p in permRows) {
                    val g = p.granted
                    val color = if (g >= 6) cBad else if (g >= 3) cWarn else cText
                    val names = p.perms.filter { it.second }.map { it.first }
                    var sub = g.toString() + " permisos sensibles concedidos de " + p.perms.size + " declarados"
                    if (names.isNotEmpty()) sub += "\n" + names.take(5).joinToString(", ") + (if (names.size > 5) "…" else "")
                    if (!p.hasInternet) sub += "\nSin permiso INTERNET"
                    rows.add(Row(p.label, sub, color, p))
                }
            }
            else -> for (l in Lessons.ALL) rows.add(Row(l.first, l.second, cText, "lesson"))
        }
        adapter.rows = rows
        adapter.notifyDataSetChanged()
    }

    private fun onRow(r: Row) {
        val tag = r.tag
        when {
            tag is Flow -> showFlow(tag)
            tag is AppStat -> dialog(tag.label, Insights.forApp(tag, Perms.forPackage(packageManager, tag.pkg)), tag.pkg)
            tag is Perms.Info -> {
                val sb = StringBuilder().append(tag.pkg).append("\n\n")
                for (p in tag.perms) sb.append(if (p.second) "✔ " else "✖ ").append(p.first).append(if (p.second) "  (concedido)\n" else "  (no concedido)\n")
                if (tag.perms.isEmpty()) sb.append("No declara permisos sensibles.\n")
                sb.append("\n").append(if (tag.hasInternet) "Tiene permiso INTERNET: puede enviar datos fuera del móvil." else "No declara INTERNET.")
                sb.append("\n\nPara conceder o retirar permisos, abre los ajustes de la app.")
                dialog(tag.label, sb.toString(), tag.pkg)
            }
            tag == "self" -> {
                if (Build.VERSION.SDK_INT >= 33) notifLauncher.launch(Manifest.permission.POST_NOTIFICATIONS)
                openSettings(packageName)
            }
            tag == "lesson" -> dialog(r.title, r.sub, null)
        }
    }

    private fun showFlow(f: Flow) {
        val sb = StringBuilder()
        sb.append("Aplicación: ").append(f.stat.label).append("\n")
        sb.append("Destino: ").append(f.dstIp).append(":").append(f.dstPort).append("\n")
        sb.append("Nombre: ").append(f.name()).append("\n")
        f.org?.let { sb.append("Propietario: ").append(it).append("\n") }
        sb.append("Protocolo: ").append(f.proto).append("\n")
        sb.append("Subido: ").append(fmt(f.up.get())).append("   Bajado: ").append(fmt(f.down.get())).append("\n")
        sb.append("Duración: ").append((f.last - f.start) / 1000).append(" s\n")
        if (f.mitm) sb.append("\n⚠ ESTA CONEXIÓN FUE DESCIFRADA (MITM TLS)\n")
        sb.append("\nLo que ve el DPI:\n")
        val notes = synchronized(f.notes) { ArrayList(f.notes) }
        if (notes.isEmpty()) sb.append("• (nada que decir de esta conexión)\n")
        for (n in notes) sb.append("• ").append(n).append("\n")
        sb.append("\nLo que NO se ve: si es TLS y no hay MITM, el contenido va cifrado.")
        dialog("Detalle de la conexión", sb.toString(), null)
    }

    private fun exportCaCert() {
        try {
            val cert = CaManager.getCaCertificate()
            val pem = "-----BEGIN CERTIFICATE-----\n" +
                Base64.encodeToString(cert.encoded, Base64.NO_WRAP) +
                "\n-----END CERTIFICATE-----\n"
            val filename = "netlens-ca.crt"
            if (Build.VERSION.SDK_INT >= 29) {
                val values = ContentValues().apply {
                    put(MediaStore.Downloads.DISPLAY_NAME, filename)
                    put(MediaStore.Downloads.MIME_TYPE, "application/x-x509-ca-cert")
                    put(MediaStore.Downloads.IS_PENDING, 1)
                }
                val uri = contentResolver.insert(MediaStore.Downloads.EXTERNAL_CONTENT_URI, values)
                if (uri != null) {
                    contentResolver.openOutputStream(uri)?.use { it.write(pem.toByteArray()) }
                    values.clear()
                    values.put(MediaStore.Downloads.IS_PENDING, 0)
                    contentResolver.update(uri, values, null, null)
                    Toast.makeText(this, "CA guardada en Descargas como $filename", Toast.LENGTH_LONG).show()
                }
            } else {
                val dir = Environment.getExternalStoragePublicDirectory(Environment.DIRECTORY_DOWNLOADS)
                val file = File(dir, filename)
                file.writeText(pem)

    private const val CA_ALIAS = "netlens-ca"
    private const val KS_FILE = "netlens-ca.p12"
    private const val KS_PASS = "netlens-ca-password"
    private const val CA_VALIDITY_DAYS = 3650L
    private const val SRV_VALIDITY_DAYS = 365L

    private lateinit var caKey: PrivateKey
    private lateinit var caCert: X509Certificate
    private var initialized = false
    private val serverCerts = ConcurrentHashMap<String, Pair<PrivateKey, X509Certificate>>()

    @Synchronized
    fun init(context: Context) {
        if (initialized) return
        try {
            val ksFile = File(context.filesDir, KS_FILE)
            val ks = KeyStore.getInstance("PKCS12")
            if (ksFile.exists()) {
                ksFile.inputStream().use { ks.load(it, KS_PASS.toCharArray()) }
                caKey = ks.getKey(CA_ALIAS, KS_PASS.toCharArray()) as PrivateKey
                caCert = ks.getCertificate(CA_ALIAS) as X509Certificate
                Log.i(TAG, "CA cargada desde $ksFile")
            } else {
                val kp = KeyPairGenerator.getInstance("RSA").apply { initialize(2048) }.generateKeyPair()
                val now = System.currentTimeMillis()
                val nb = Date(now - TimeUnit.DAYS.toMillis(1))
                val na = Date(now + TimeUnit.DAYS.toMillis(CA_VALIDITY_DAYS))
                val subject = X500Name("CN=NetLens CA, O=NetLens Lab, C=ES")
                val builder = JcaX509v3CertificateBuilder(subject, BigInteger.valueOf(now), nb, na, subject, kp.public)
                builder.addExtension(Extension.basicConstraints, true, BasicConstraints(true))
                builder.addExtension(Extension.keyUsage, true, KeyUsage(KeyUsage.keyCertSign or KeyUsage.cRLSign))
                val signer = JcaContentSignerBuilder("SHA256withRSA").build(kp.private)
                caCert = JcaX509CertificateConverter().getCertificate(builder.build(signer))
                caKey = kp.private
                ks.load(null, null)
                ks.setKeyEntry(CA_ALIAS, caKey, KS_PASS.toCharArray(), arrayOf(caCert))
                ksFile.outputStream().use { ks.store(it, KS_PASS.toCharArray()) }
                Log.i(TAG, "CA generada y guardada en $ksFile")
            }
            initialized = true
        } catch (e: Exception) {
            Log.e(TAG, "Error inicializando CA", e)
        }
    }

    fun isInstalled(context: Context): Boolean {
        // Comprobación simple: si la CA ya se exportó, consideramos que puede estar instalada
        return File(context.filesDir, KS_FILE).exists()
    }

    fun getCaCertificate(): X509Certificate = caCert

    fun fingerprint(): String {
        val md = MessageDigest.getInstance("SHA-256")
        val d = md.digest(caCert.encoded)
        return d.joinToString(":") { String.format("%02X", it) }
    }

    fun getServerCert(domain: String): Pair<PrivateKey, X509Certificate> {
        return serverCerts.getOrPut(domain) { generateServerCert(domain) }
    }

    private fun generateServerCert(domain: String): Pair<PrivateKey, X509Certificate> {
        val kp = KeyPairGenerator.getInstance("RSA").apply { initialize(2048) }.generateKeyPair()
        val now = System.currentTimeMillis()
        val nb = Date(now - TimeUnit.DAYS.toMillis(1))
        val na = Date(now + TimeUnit.DAYS.toMillis(SRV_VALIDITY_DAYS))
        val subject = X500Name("CN=$domain")
        val issuer = X500Name(caCert.subjectX500Principal.name)
        val builder = JcaX509v3CertificateBuilder(issuer, BigInteger.valueOf(now + domain.hashCode()), nb, na, subject, kp.public)
        builder.addExtension(Extension.basicConstraints, false, BasicConstraints(false))
        builder.addExtension(Extension.subjectAlternativeName, false, GeneralNames(GeneralName(GeneralName.dNSName, domain)))
        val signer = JcaContentSignerBuilder("SHA256withRSA").build(caKey)
        val cert = JcaX509CertificateConverter().getCertificate(builder.build(signer))
        return Pair(kp.private, cert)
    }

object Dpi {
    class TlsInfo(val sni: String?, val alpn: List<String>, val tls13: Boolean, val ech: Boolean, val complete: Boolean, val legacyVer: Int)
    class HttpInfo(val method: String, val host: String?, val path: String, val queryKeys: List<String>, val userAgent: String?)

    private fun u8(b: ByteArray, p: Int): Int = b[p].toInt() and 0xFF
    private fun u16(b: ByteArray, p: Int): Int = (u8(b, p) shl 8) or u8(b, p + 1)

    fun tls(b: ByteArray): TlsInfo? {
        if (b.size < 6 || u8(b, 0) != 0x16 || u8(b, 1) != 3) return null
        if (u8(b, 5) != 1) return null
        var sni: String? = null
        val alpn = ArrayList<String>()
        var t13 = false; var ech = false; var complete = false; var ver = 0
        try {
            var p = 9
            ver = u16(b, p); p += 2 + 32
            p += 1 + u8(b, p); p += 2 + u16(b, p); p += 1 + u8(b, p)
            val extLen = u16(b, p); p += 2
            val end = p + extLen
            while (p + 4 <= end) {
                val type = u16(b, p); val len = u16(b, p + 2); val q = p + 4
                if (type == 0 && len >= 5) { val nl = u16(b, q + 3); sni = String(b, q + 5, nl, Charsets.US_ASCII) }
                else if (type == 16) { var a = q + 2; val ae = q + 2 + u16(b, q); while (a < ae) { val l = u8(b, a); alpn.add(String(b, a + 1, l, Charsets.US_ASCII)); a += 1 + l } }
                else if (type == 43) { var a = q + 1; val ae = q + 1 + u8(b, q); while (a + 1 < ae) { if (u16(b, a) == 0x0304) t13 = true; a += 2 } }
                else if (type == 0xFE0D) ech = true
                p = q + len
            }
            complete = true
        } catch (e: Exception) { complete = false }
        return TlsInfo(sni, alpn, t13, ech, complete, ver)
    }

    private val METHODS = listOf("GET ", "POST ", "PUT ", "HEAD ", "DELETE ", "OPTIONS ", "PATCH ", "CONNECT ")
    private val HOST_RE = Regex("(?im)^host:\\s*([^\\r\\n]+)")
    private val UA_RE = Regex("(?im)^user-agent:\\s*([^\\r\\n]+)")

    fun http(b: ByteArray): HttpInfo? {
        val head = String(b, 0, minOf(b.size, 2048), Charsets.ISO_8859_1)
        val m = METHODS.firstOrNull { head.startsWith(it) } ?: return null
        var firstEnd = head.indexOf('\r'); if (firstEnd < 0) firstEnd = head.indexOf('\n')
        val line = if (firstEnd > 0) head.substring(0, firstEnd) else head
        val parts = line.split(' ')
        val target = if (parts.size > 1) parts[1] else ""
        val path = target.substringBefore('?')
        val keys = if (target.contains('?')) target.substringAfter('?').split('&').map { it.substringBefore('=') }.filter { it.isNotEmpty() }.take(12) else emptyList()
        val host = HOST_RE.find(head)?.groupValues?.get(1)?.trim()
        val ua = UA_RE.find(head)?.groupValues?.get(1)?.trim()
        return HttpInfo(m.trim(), host, path.take(80), keys, ua?.take(80))
    }

    private val HINTS = listOf("password=", "passwd=", "pwd=", "token=", "api_key=", "apikey=", "access_token=", "authorization:", "email=", "mail=", "imei=", "android_id=", "advertising_id=", "gaid=", "aaid=", "lat=", "lon=", "latitude=", "longitude=", "device_id=", "deviceid=", "phone=")

    }

    private fun readName(b: ByteArray, start: Int): Pair<String, Int> {
        val sb = StringBuilder(); var p = start; var next = -1; var guard = 0
        while (true) {
            guard++; if (guard > 60) throw IllegalStateException("dns loop")
            val l = u8(b, p)
            if (l == 0) { p++; break }
            if ((l and 0xC0) == 0xC0) { if (next < 0) next = p + 2; p = ((l and 0x3F) shl 8) or u8(b, p + 1); continue }
            if (sb.isNotEmpty()) sb.append('.')
            sb.append(String(b, p + 1, l, Charsets.US_ASCII)); p += 1 + l
        }
        return Pair(sb.toString(), if (next >= 0) next else p)
    }

    fun dnsQuery(b: ByteArray): String? = try { if (b.size < 13) null else readName(b, 12).first } catch (e: Exception) { null }

    fun dnsAnswers(b: ByteArray, n: Int): Pair<String, List<String>>? {
        val d = b.copyOf(n); val ips = ArrayList<String>(); var qname = ""
        try {
            val qd = u16(d, 4); val an = u16(d, 6); var p = 12

                p += rdlen
            }
        } catch (e: Exception) {}
        return if (ips.isEmpty()) null else Pair(qname, ips)
    }

        if (a.size >= 8) {

        }

object Pkt {
    const val FIN = 1; const val SYN = 2; const val RST = 4; const val PSH = 8; const val ACK = 16
    private val ipId = AtomicInteger(1)
    fun u8(b: ByteArray, p: Int): Int = b[p].toInt() and 0xFF
    fun u16(b: ByteArray, p: Int): Int = (u8(b, p) shl 8) or u8(b, p + 1)
    fun u32(b: ByteArray, p: Int): Int = (u8(b, p) shl 24) or (u8(b, p + 1) shl 16) or (u8(b, p + 2) shl 8) or u8(b, p + 3)
    fun put16(b: ByteArray, p: Int, v: Int) { b[p] = (v shr 8).toByte(); b[p + 1] = v.toByte() }
    fun put32(b: ByteArray, p: Int, v: Int) { b[p] = (v shr 24).toByte(); b[p + 1] = (v shr 16).toByte(); b[p + 2] = (v shr 8).toByte(); b[p + 3] = v.toByte() }
    fun ipStr(b: ByteArray, p: Int): String = "" + u8(b, p) + "." + u8(b, p + 1) + "." + u8(b, p + 2) + "." + u8(b, p + 3)
    fun checksum(b: ByteArray, off: Int, len: Int, init: Long): Int {
        var sum: Long = init; var i = off; val end = off + len
        while (i + 1 < end) { sum += ((u8(b, i) shl 8) or u8(b, i + 1)).toLong(); i += 2 }
        if (i < end) sum += (u8(b, i) shl 8).toLong()
        while ((sum shr 16) != 0L) sum = (sum and 0xFFFFL) + (sum shr 16)
        return (sum.inv() and 0xFFFFL).toInt()
    }
    private fun pseudo(src: ByteArray, dst: ByteArray, proto: Int, len: Int): Long {
        var s = 0L
        s += ((u8(src, 0) shl 8) or u8(src, 1)).toLong(); s += ((u8(src, 2) shl 8) or u8(src, 3)).toLong()
        s += ((u8(dst, 0) shl 8) or u8(dst, 1)).toLong(); s += ((u8(dst, 2) shl 8) or u8(dst, 3)).toLong()
        s += proto.toLong(); s += len.toLong(); return s
    }
    private fun ipHeader(p: ByteArray, total: Int, proto: Int, src: ByteArray, dst: ByteArray) {
        p[0] = 0x45.toByte(); p[1] = 0
        put16(p, 2, total); put16(p, 4, ipId.getAndIncrement() and 0xFFFF); put16(p, 6, 0x4000)
        p[8] = 64; p[9] = proto.toByte()
        System.arraycopy(src, 0, p, 12, 4); System.arraycopy(dst, 0, p, 16, 4)
        put16(p, 10, 0); put16(p, 10, checksum(p, 0, 20, 0L))
    }
    fun tcp(src: ByteArray, dst: ByteArray, sp: Int, dp: Int, seq: Int, ack: Int, flags: Int, win: Int, data: ByteArray?, off: Int, len: Int, mss: Int): ByteArray {
        val th = if (mss > 0) 24 else 20; val total = 20 + th + len
        val p = ByteArray(total); ipHeader(p, total, 6, src, dst)
        val t = 20
        put16(p, t, sp); put16(p, t + 2, dp); put32(p, t + 4, seq); put32(p, t + 8, ack)
        p[t + 12] = ((th / 4) shl 4).toByte(); p[t + 13] = flags.toByte(); put16(p, t + 14, win)
        if (mss > 0) { p[t + 20] = 2; p[t + 21] = 4; put16(p, t + 22, mss) }
        if (len > 0 && data != null) System.arraycopy(data, off, p, t + th, len)
        put16(p, t + 16, checksum(p, t, th + len, pseudo(src, dst, 6, th + len)))
        return p
    }
    fun udp(src: ByteArray, dst: ByteArray, sp: Int, dp: Int, data: ByteArray, off: Int, len: Int): ByteArray {
        val total = 20 + 8 + len; val p = ByteArray(total); ipHeader(p, total, 17, src, dst)
        val t = 20; put16(p, t, sp); put16(p, t + 2, dp); put16(p, t + 4, 8 + len)
        System.arraycopy(data, off, p, t + 8, len)
        var c = checksum(p, t, 8 + len, pseudo(src, dst, 17, 8 + len)); if (c == 0) c = 0xFFFF
        put16(p, t + 6, c); return p
    }
}

class TcpConn(
    private val eng: Engine, private val key: String,
    private val cIp: ByteArray, private val sIp: ByteArray,
    private val cPort: Int, private val sPort: Int,
    clientIsn: Int, val flow: Flow
) {
    companion object { val FIN_MARK = ByteArray(0); const val MSS = 1400 }
    private val lk = ReentrantLock(); private val cond = lk.newCondition()
    private val iss: Int = Random().nextInt()
    private var rcvNxt: Int = clientIsn + 1; private var sndNxt: Int = iss
    private var lastAck: Int = iss; private var cWnd: Int = 65535
    private var established = false; private var finFromClient = false; private var finToClient = false
    @Volatile private var closed = false; @Volatile private var zeroWin = false
    @Volatile var lastActive: Long = System.currentTimeMillis()
    private var sock: Socket? = null
    private val wq = LinkedBlockingQueue<ByteArray>()
    private val queued = AtomicInteger()
    private val dpiBuf = ByteArrayOutputStream()
    private var dpiDone = false

    fun begin() {
        eng.pool.execute {
            try {
                val s = Socket(); eng.protect(s); s.tcpNoDelay = true
                if (Store.mitmEnabled && sPort == 443) {
                    s.connect(InetSocketAddress(InetAddress.getByName("127.0.0.1"), MitmProxy.PORT), 8000)
                } else {
                    s.connect(InetSocketAddress(InetAddress.getByAddress(sIp), sPort), 8000)
                }
                sock = s
                lk.withLock {
                    eng.write(Pkt.tcp(sIp, cIp, sPort, cPort, iss, rcvNxt, Pkt.SYN or Pkt.ACK, 65535, null, 0, 0, MSS))
                    sndNxt = iss + 1; lastAck = iss + 1; established = true
                }
                eng.pool.execute { writerLoop(s) }
                readerLoop(s)
            } catch (e: Exception) {
                if (!closed) { flow.notes.add("✖ No se pudo conectar: " + (e.message ?: e.javaClass.simpleName)); lk.withLock { rst() } }
                teardown()
            }
        }
    }

    private fun advWin(): Int { val q = queued.get(); if (q > 262144) { zeroWin = true; return 0 }; return 65535 }
    private fun sendAck() { eng.write(Pkt.tcp(sIp, cIp, sPort, cPort, sndNxt, rcvNxt, Pkt.ACK, advWin(), null, 0, 0, 0)) }
    private fun rst() { val seq = if (established) sndNxt else 0; eng.write(Pkt.tcp(sIp, cIp, sPort, cPort, seq, rcvNxt, Pkt.RST or Pkt.ACK, 0, null, 0, 0, 0)) }

    fun onSegment(flags: Int, seq: Int, ack: Int, win: Int, data: ByteArray?) {
        lastActive = System.currentTimeMillis()
        if ((flags and Pkt.RST) != 0) { teardown(); return }
        lk.withLock {
            if (!established) return
            if ((flags and Pkt.SYN) != 0) { eng.write(Pkt.tcp(sIp, cIp, sPort, cPort, iss, rcvNxt, Pkt.SYN or Pkt.ACK, 65535, null, 0, 0, MSS)); return }
            if ((flags and Pkt.ACK) != 0) { if ((ack - lastAck) > 0) lastAck = ack; cWnd = win; cond.signalAll() }
            val dlen = if (data == null) 0 else data.size
            if (dlen > 0 && data != null) {
                if (seq == rcvNxt) { rcvNxt += dlen; wq.add(data); queued.addAndGet(dlen); flow.addUp(dlen); inspect(data) }
                sendAck()
            }
            if ((flags and Pkt.FIN) != 0) { if (!finFromClient && (seq + dlen) == rcvNxt) { finFromClient = true; rcvNxt += 1; wq.add(FIN_MARK) }; sendAck() }
            checkDone()
        }
    }

    private fun inspect(d: ByteArray) {
        if (dpiDone) return
        dpiBuf.write(d, 0, d.size)
        if (dpiBuf.size() > 16384) { dpiDone = true; return }
        val a = dpiBuf.toByteArray()
        dpiDone = Dpi.inspectOut(flow, a, sPort)
    }

    private fun checkDone() { if (finFromClient && finToClient) eng.timer.schedule(Runnable { teardown() }, 3, TimeUnit.SECONDS) }

    private fun writerLoop(s: Socket) {
        try {
            val os = s.getOutputStream()
            while (!closed) {
                val d = wq.poll(500, TimeUnit.MILLISECONDS) ?: continue
                if (d === FIN_MARK) { try { s.shutdownOutput() } catch (e: Exception) {}; continue }
                os.write(d); os.flush()
                val q = queued.addAndGet(-d.size)
                if (zeroWin && q < 131072) lk.withLock { zeroWin = false; sendAck() }
                lastActive = System.currentTimeMillis()
            }
        } catch (e: Exception) { if (!closed) { lk.withLock { rst() }; teardown() } }
    }

    private fun readerLoop(s: Socket) {
        val buf = ByteArray(MSS)
        try {
            val ins = s.getInputStream()
            while (!closed) {
                val n = ins.read(buf)
                if (n < 0) break
                if (n == 0) continue
                lk.withLock {
                    while (!closed && (sndNxt - lastAck) > 0 && (sndNxt - lastAck) + n > cWnd) cond.await(200, TimeUnit.MILLISECONDS)
                    if (!closed) { eng.write(Pkt.tcp(sIp, cIp, sPort, cPort, sndNxt, rcvNxt, Pkt.ACK or Pkt.PSH, advWin(), buf, 0, n, 0)); sndNxt += n }
                }
                flow.addDown(n); lastActive = System.currentTimeMillis()
            }
            lk.withLock {
                if (!closed) { eng.write(Pkt.tcp(sIp, cIp, sPort, cPort, sndNxt, rcvNxt, Pkt.FIN or Pkt.ACK, advWin(), null, 0, 0, 0)); sndNxt += 1; finToClient = true; checkDone() }
            }
        } catch (e: Exception) { if (!closed) { lk.withLock { rst() }; teardown() } }
    }

    fun teardown() {
        if (closed) return
        closed = true
        try { sock?.close() } catch (e: Exception) {}
        flow.closed = true; eng.removeTcp(key); lk.withLock { cond.signalAll() }
    }
}

class Engine(
    private val svc: VpnService, private val out: FileOutputStream,
    private val pm: PackageManager, private val cm: ConnectivityManager
) {
    val pool = Executors.newCachedThreadPool()
    val timer = Executors.newScheduledThreadPool(1)
    private val tcp = ConcurrentHashMap<String, TcpConn>()
    private val udp = ConcurrentHashMap<String, UdpCtx>()
    private val uidCache = ConcurrentHashMap<Int, AppStat>()
    @Volatile private var alive = true

    init { timer.scheduleWithFixedDelay({ janitor() }, 20, 20, TimeUnit.SECONDS) }
    fun protect(s: Socket) { svc.protect(s) }
    fun protect(s: DatagramSocket) { svc.protect(s) }
    fun write(p: ByteArray) { synchronized(out) { try { out.write(p) } catch (e: Exception) {} } }
    fun removeTcp(key: String) { tcp.remove(key) }

    private fun janitor() {
        val now = System.currentTimeMillis()
        for (c in tcp.values) if (now - c.lastActive > 180000) c.teardown()
        for (e in udp.entries) if (e.value.sock == null && now - e.value.flow.last > 60000) udp.remove(e.key)
    }

    fun stop() {
        alive = false
        for (c in tcp.values) c.teardown()
        for (u in udp.values) { try { u.sock?.close() } catch (e: Exception) {} }
        pool.shutdownNow(); timer.shutdownNow()
    }

    fun handle(b: ByteArray, n: Int) {
        if (n < 20) return
        if (((b[0].toInt() shr 4) and 0xF) != 4) return
        val ihl = (Pkt.u8(b, 0) and 0xF) * 4
        if (ihl < 20 || n < ihl) return
        val total = Pkt.u16(b, 2)
        val len = if (total in 20..n) total else n
        if ((Pkt.u16(b, 6) and 0x3FFF) != 0) return
        when (Pkt.u8(b, 9)) { 6 -> handleTcp(b, len, ihl); 17 -> handleUdp(b, len, ihl) }
    }

    private fun ownerUid(proto: Int, src: ByteArray, sp: Int, dst: ByteArray, dp: Int): Int {
        if (Build.VERSION.SDK_INT < 29) return -1
        return try { cm.getConnectionOwnerUid(proto, InetSocketAddress(InetAddress.getByAddress(src), sp), InetSocketAddress(InetAddress.getByAddress(dst), dp)) } catch (e: Exception) { -1 }
    }

    private fun appFor(proto: Int, src: ByteArray, sp: Int, dst: ByteArray, dp: Int): AppStat {
        val uid = ownerUid(proto, src, sp, dst, dp)
        return uidCache.getOrPut(uid) {
            val pkgs = if (uid >= 0) pm.getPackagesForUid(uid) else null
            val pkg = if (pkgs != null && pkgs.isNotEmpty()) pkgs[0] else if (uid == 1000) "android" else "uid:" + uid
            val label = try { pm.getApplicationInfo(pkg, 0).loadLabel(pm).toString() }
            catch (e: Exception) { if (uid == 1000) "Sistema Android" else if (uid < 0) "App desconocida" else pkg }
            Store.app(pkg, label)
        }
    }

    private fun handleTcp(b: ByteArray, len: Int, ihl: Int) {
        if (len < ihl + 20) return
        val t = ihl
        val sp = Pkt.u16(b, t); val dp = Pkt.u16(b, t + 2)
        val seq = Pkt.u32(b, t + 4); val ack = Pkt.u32(b, t + 8)
        val doff = (Pkt.u8(b, t + 12) shr 4) * 4
        val flags = Pkt.u8(b, t + 13); val win = Pkt.u16(b, t + 14)
        if (doff < 20 || len < t + doff) return
        val dlen = len - t - doff
        val key = "" + sp + ">" + Pkt.ipStr(b, 16) + ":" + dp
        val c = tcp[key]
        val isSyn = (flags and Pkt.SYN) != 0; val isAck = (flags and Pkt.ACK) != 0; val isRst = (flags and Pkt.RST) != 0
        if (c == null) {
            if (isSyn && !isAck) {
                val src = b.copyOfRange(12, 16); val dst = b.copyOfRange(16, 20)
                val stat = appFor(6, src, sp, dst, dp)
                val flow = Store.newFlow(stat, "TCP", Pkt.ipStr(b, 16), dp)
                val deny = Protection.reason(stat.pkg, Store.dns[Pkt.ipStr(b, 16)])
                if (deny != null) {
                    flow.notes.add("Bloqueado: " + deny); flow.closed = true; Protection.record(stat.pkg)
                    write(Pkt.tcp(dst, src, dp, sp, 0, seq + 1, Pkt.RST or Pkt.ACK, 0, null, 0, 0, 0)); return
                }
                val conn = TcpConn(this, key, src, dst, sp, dp, seq, flow)
                tcp[key] = conn; conn.begin()
            } else if (!isRst) {
                val src = b.copyOfRange(16, 20); val dst = b.copyOfRange(12, 16)
                if (isAck) write(Pkt.tcp(src, dst, dp, sp, ack, 0, Pkt.RST, 0, null, 0, 0, 0))
                else {
                    val add = dlen + (if (isSyn) 1 else 0) + (if ((flags and Pkt.FIN) != 0) 1 else 0)
                    write(Pkt.tcp(src, dst, dp, sp, 0, seq + add, Pkt.RST or Pkt.ACK, 0, null, 0, 0, 0))
                }
            }
            return
        }
        val data = if (dlen > 0) b.copyOfRange(t + doff, t + doff + dlen) else null
        c.onSegment(flags, seq, ack, win, data)
    }

    private fun handleUdp(b: ByteArray, len: Int, ihl: Int) {
        if (len < ihl + 8) return
        val sp = Pkt.u16(b, ihl); val dp = Pkt.u16(b, ihl + 2)
        val ulen = Pkt.u16(b, ihl + 4)
        val dlen = minOf(ulen - 8, len - ihl - 8)
        if (dlen < 0) return
        if (Pkt.u8(b, 16) >= 224) return
        val key = "" + sp + ">" + Pkt.ipStr(b, 16) + ":" + dp
        val src = b.copyOfRange(12, 16); val dst = b.copyOfRange(16, 20)
        val ctx: UdpCtx = udp[key] ?: (createUdp(key, src, dst, sp, dp, Pkt.ipStr(b, 16)) ?: return)
        val sock = ctx.sock
        if (sock == null) { ctx.flow.last = System.currentTimeMillis(); return }
        val payload = b.copyOfRange(ihl + 8, ihl + 8 + dlen)
        val host = if (dp == 53) Dpi.dnsQuery(payload) else Store.dns[Pkt.ipStr(b, 16)]
        val deny = Protection.reason(ctx.flow.stat.pkg, host)
        if (deny != null) {
            Protection.record(ctx.flow.stat.pkg); ctx.flow.notes.add("Bloqueado: " + deny)
            if (dp == 53 && payload.size >= 12) {
                val reply = payload.copyOf()
                reply[2] = ((reply[2].toInt() and 1) or 0x80).toByte()
                reply[3] = 0x83.toByte()
                for (p in 6..11) reply[p] = 0
                write(Pkt.udp(dst, src, dp, sp, reply, 0, reply.size))
            }
            return
        }

        if (dp == 53) { val q = Dpi.dnsQuery(payload); if (q != null) { ctx.flow.noteHost(q); ctx.flow.notes.add("DNS: la app pregunta por " + q) } }
        try { sock.send(DatagramPacket(payload, payload.size)); ctx.flow.addUp(dlen) } catch (e: Exception) {}
    }

    private fun createUdp(key: String, src: ByteArray, dst: ByteArray, sp: Int, dp: Int, dstStr: String): UdpCtx? {
        val stat = appFor(17, src, sp, dst, dp)
        val proto = if (dp == 53) "DNS" else if (dp == 443) "UDP·QUIC" else "UDP"
        val flow = Store.newFlow(stat, proto, dstStr, dp)
        if (dp == 443 && Store.blockQuic) { flow.notes.add("⛔ QUIC bloqueado: la app caerá a TCP+TLS"); flow.closed = true; udp[key] = UdpCtx(null, flow); return udp[key] }
        val c: UdpCtx = try {
            val s = DatagramSocket(); protect(s); s.connect(InetAddress.getByAddress(dst), dp)
            s.soTimeout = if (dp == 53) 5000 else 30000
            UdpCtx(s, flow)
        } catch (e: Exception) { flow.notes.add("✖ No se pudo abrir el socket UDP"); flow.closed = true; return null }
        udp[key] = c
        if (dp == 443) flow.notes.add("QUIC/HTTP3: va cifrado. Activa «Bloquear QUIC» para forzar TCP+TLS")
        pool.execute { udpReceiver(key, c, src, dst, sp, dp) }
        return c
    }

    private fun udpReceiver(key: String, ctx: UdpCtx, src: ByteArray, dst: ByteArray, sp: Int, dp: Int) {
        val s = ctx.sock ?: return
        val buf = ByteArray(4096)
        try {
            while (alive) {
                val dpk = DatagramPacket(buf, buf.size); s.receive(dpk)
                val n = dpk.length; ctx.flow.addDown(n)
                if (dp == 53) { val r = Dpi.dnsAnswers(buf, n); if (r != null) { for (ip in r.second) Store.dns[ip] = r.first; ctx.flow.notes.add("DNS respuesta: " + r.first + " → " + r.second.joinToString(", ")) } }
                if (n <= 1472) write(Pkt.udp(dst, src, dp, sp, buf, 0, n))
                if (dp == 53) break
            }
        } catch (e: SocketTimeoutException) {} catch (e: Exception) {} finally {
            try { s.close() } catch (e: Exception) {}
            ctx.flow.closed = true; udp.remove(key)
        }
    }
}

object Export {
    private fun q(s: String): String {
        var v = s.replace("\r", " ").replace("\n", " | ")
        if (v.isNotEmpty() && (v[0] == '=' || v[0] == '+' || v[0] == '-' || v[0] == '@')) v = "'" + v
        return "\"" + v.replace("\"", "\"\"") + "\""
    }
    private fun baseDomain(h: String): String {
        if (h.isEmpty()) return ""
        if (h.all { it.isDigit() || it == '.' }) return "(IP)"
        val p = h.split('.')
        return if (p.size <= 2) h else p[p.size - 2] + "." + p[p.size - 1]
    }
    fun build(flows: List<Flow>, raw: Boolean): String {
        val sb = StringBuilder("\uFEFF")
        val full = SimpleDateFormat("yyyy-MM-dd HH:mm:ss", Locale.US)
        val minute = SimpleDateFormat("yyyy-MM-dd HH:mm", Locale.US)
        if (raw) sb.append("inicio,app,paquete,protocolo,ip_destino,puerto,nombre,categoria,bytes_subidos,bytes_bajados,duracion_s,mitm,notas\n")
        else sb.append("minuto,app,protocolo,puerto,dominio_base,categoria,subida_aprox_KB,bajada_aprox_KB,http_sin_cifrar,tls,mitm\n")
        for (f in flows) {
            val notes = synchronized(f.notes) { ArrayList(f.notes) }
            val name = f.name(); val cat = Dpi.category(name) ?: ""
            if (raw) {
                sb.append(q(full.format(Date(f.start)))).append(',').append(q(f.stat.label)).append(',').append(q(f.stat.pkg)).append(',')
                    .append(q(f.proto)).append(',').append(q(f.dstIp)).append(',').append(f.dstPort).append(',').append(q(name)).append(',')

                    .append(if (f.mitm) "si" else "no").append(',').append(q(notes.joinToString(" | "))).append('\n')
            } else {
                sb.append(q(minute.format(Date(f.start)))).append(',').append(q(f.stat.label)).append(',').append(q(f.proto)).append(',')
                    .append(f.dstPort).append(',').append(q(baseDomain(name))).append(',').append(q(cat)).append(',')

            }
        }

    }

class MitmProxy(
    private val port: Int = PORT,
    private val protector: (Socket) -> Boolean
) {
    companion object {
        const val PORT = 8888
        private const val TAG = "MitmProxy"
    }

    private var serverSocket: ServerSocket? = null
    @Volatile private var running = false
    private lateinit var sslContext: SSLContext

    fun start() {
        if (!Security.getProviders().any { it.name == BouncyCastleProvider.PROVIDER_NAME }) {
            Security.addProvider(BouncyCastleProvider())
        }
        sslContext = buildSslContext()
        running = true
        Thread({
            try {
                serverSocket = ServerSocket(port, 50, java.net.InetAddress.getByName("127.0.0.1"))
                Log.i(TAG, "MITM escuchando en 127.0.0.1:$port")
                while (running) {
                    val client = serverSocket!!.accept()
                    Thread { handleClient(client) }.start()
                }
            } catch (e: Exception) {
                if (running) Log.e(TAG, "Error en accept", e)
            }
        }, "mitm-accept").start()
    }

    fun stop() {
        running = false
        try { serverSocket?.close() } catch (e: Exception) {}
    }

    private fun buildSslContext(): SSLContext {
        val km = DynamicSniKeyManager { domain -> CaManager.getServerCert(domain) }
        val ctx = SSLContext.getInstance("TLS")
        ctx.init(arrayOf<KeyManager>(km), null, null)
        return ctx
    }

    private fun handleClient(client: Socket) {
        var sslClient: SSLSocket? = null
        try {
            client.soTimeout = 20000
            val sslFactory: SSLSocketFactory = sslContext.socketFactory as SSLSocketFactory
            sslClient = sslFactory.createSocket(client, null, client.port, false) as SSLSocket
            sslClient.useClientMode = false
            sslClient.soTimeout = 20000
            sslClient.startHandshake()

            val sni = try {
                sslClient.handshakeSession?.peerHost
            } catch (e: Exception) {
                null
            }
            if (sni.isNullOrEmpty()) {
                Log.w(TAG, "SNI vacío, cerrando")
                try { sslClient.close() } catch (_: Exception) {}
                return
            }
            Log.i(TAG, "Cliente TLS OK para $sni")
            handleTlsMitm(sslClient, sni)
        } catch (e: Exception) {
            Log.e(TAG, "handleClient: ${e.javaClass.simpleName}: ${e.message}")
            try { sslClient?.close() } catch (_: Exception) {}
            try { client.close() } catch (_: Exception) {}
        }
    }

    private fun handleTlsMitm(sslClient: SSLSocket, sni: String) {
        var sslRemote: SSLSocket? = null
        try {
            val remote = Socket()
            protector(remote)
            remote.connect(InetSocketAddress(sni, 443), 10000)
            val defaultFactory: SSLSocketFactory = SSLSocketFactory.getDefault() as SSLSocketFactory
            sslRemote = defaultFactory.createSocket(remote, sni, 443, false) as SSLSocket
            sslRemote.startHandshake()
            Log.i(TAG, "Remoto TLS OK para $sni")

            markMitm(sni)

            val in1 = sslClient.inputStream
            val out1 = sslRemote.outputStream
            val in2 = sslRemote.inputStream
            val out2 = sslClient.outputStream

            val t1 = Thread { pipe(in1, out1) }
            val t2 = Thread { pipe(in2, out2) }
            t1.start(); t2.start()
            t1.join(180000); t2.join(180000)
        } catch (e: Exception) {
            Log.w(TAG, "MITM falló para $sni: ${e.javaClass.simpleName}: ${e.message}")
        } finally {
            try { sslClient.close() } catch (_: Exception) {}
            try { sslRemote?.close() } catch (_: Exception) {}
        }
    }

    private fun markMitm(sni: String) {
        try {
            for (f in Store.snapshot()) {
                if (f.host == sni && !f.mitm) {
                    f.mitm = true
                    f.stat.mitmCount.incrementAndGet()
                    f.notes.add("🔓 MITM TLS: contenido descifrado con CA local")
                    break
                }
            }
        } catch (e: Exception) {}
    }

    private fun pipe(input: InputStream, output: OutputStream) {
        try {
            val buf = ByteArray(8192)
            while (running) {
                val n = input.read(buf)
                if (n < 0) break
                if (n > 0) {
                    output.write(buf, 0, n)
                    output.flush()
                }
            }
        } catch (e: Exception) {
        } finally {
            try { output.close() } catch (_: Exception) {}
            try { input.close() } catch (_: Exception) {}
        }
    }
}

class DynamicSniKeyManager(
    private val certProvider: (String) -> Pair<PrivateKey, X509Certificate>
) : X509ExtendedKeyManager() {

    private val store: KeyStore = KeyStore.getInstance("PKCS12").apply { load(null, null) }
    private val pass = "cert".toCharArray()
    private val domainToAlias = ConcurrentHashMap<String, String>()
    private var counter = 0

    override fun getCertificateChain(alias: String?): Array<X509Certificate>? {
        if (alias == null) return null
        return try { store.getCertificateChain(alias) as Array<X509Certificate>? } catch (e: Exception) { null }
    }

    override fun getPrivateKey(alias: String?): PrivateKey? {
        if (alias == null) return null
        return try { store.getKey(alias, pass) as PrivateKey? } catch (e: Exception) { null }
    }

    override fun chooseServerAlias(keyType: String?, issuers: Array<out Principal>?, socket: Socket?): String? {
        val sslSock = socket as? SSLSocket ?: return null
        val domain = try {
            sslSock.handshakeSession?.peerHost ?: sslSock.session?.peerHost
        } catch (e: Exception) {
            null
        }
        if (domain.isNullOrEmpty()) return null
        return getOrCreate(domain)
    }

    private fun getOrCreate(domain: String): String? {
        domainToAlias[domain]?.let { return it }
        return try {
            val (key, cert) = certProvider(domain)
            val alias = "d${counter++}-${domain.hashCode()}"
            store.setKeyEntry(alias, key, pass, arrayOf(cert))
            domainToAlias[domain] = alias
            alias
        } catch (e: Exception) {
            null
        }
    }

    override fun getServerAliases(keyType: String?, issuers: Array<out Principal>?): Array<String>? = null
    override fun chooseClientAlias(keyType: Array<out String>?, issuers: Array<out Principal>?, socket: Socket?): String? = null
    override fun getClientAliases(keyType: String?, issuers: Array<out Principal>?): Array<String>? = null
}

import androidx.core.app.NotificationCompat
import java.io.FileInputStream
import java.io.FileOutputStream

class NetLensVpnService : VpnService() {
    companion object {
        const val ACTION_START = "com.lab.netlens.START"
        const val ACTION_STOP = "com.lab.netlens.STOP"
        const val EXTRA_PKG = "pkg"
        private const val CHANNEL = "netlens"
    }

    private var pfd: ParcelFileDescriptor? = null
    private var engine: Engine? = null
    private var proxy: MitmProxy? = null
    private var reader: Thread? = null

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        if (intent != null && intent.action == ACTION_STOP) { stopVpn(); return START_NOT_STICKY }
        startFg()
        startVpn(intent?.getStringExtra(EXTRA_PKG))
        return START_NOT_STICKY
    }

    private fun startFg() {
        val nm = getSystemService(NotificationManager::class.java)
        if (Build.VERSION.SDK_INT >= 26) nm.createNotificationChannel(NotificationChannel(CHANNEL, "Captura de red", NotificationManager.IMPORTANCE_LOW))
        val stop = PendingIntent.getService(this, 1, Intent(this, NetLensVpnService::class.java).setAction(ACTION_STOP), PendingIntent.FLAG_IMMUTABLE)
        val open = PendingIntent.getActivity(this, 2, Intent(this, MainActivity::class.java), PendingIntent.FLAG_IMMUTABLE)
        val n = NotificationCompat.Builder(this, CHANNEL)
            .setSmallIcon(android.R.drawable.ic_dialog_info)
            .setContentTitle("NetLens está observando el tráfico de este móvil")
            .setContentText("Todo se analiza en local. Toca para abrir.")
            .setContentIntent(open)
            .addAction(0, "Detener", stop)
            .setOngoing(true)
            .build()
        if (Build.VERSION.SDK_INT >= 34) startForeground(1, n, ServiceInfo.FOREGROUND_SERVICE_TYPE_SPECIAL_USE)
        else startForeground(1, n)
    }

    private fun startVpn(target: String?) {
        Protection.init(this)
        if (pfd != null) return
        val b = Builder()
            .setSession("NetLens")
            .addAddress("10.215.173.1", 32)
            .addRoute("0.0.0.0", 0)
            .addDnsServer("8.8.8.8")
            .addDnsServer("1.1.1.1")
            .setMtu(1500)
            .setBlocking(true)
        try {
            if (target != null && target.isNotEmpty()) b.addAllowedApplication(target)
            else b.addDisallowedApplication(packageName)
        } catch (e: Exception) {}
        val fd = b.establish() ?: run { stopVpn(); return }
        pfd = fd
        val out = FileOutputStream(fd.fileDescriptor)
        val inp = FileInputStream(fd.fileDescriptor)
        val eng = Engine(this, out, packageManager, getSystemService(ConnectivityManager::class.java))
        engine = eng

        if (Store.mitmEnabled) {
            try {
            val p = MitmProxy(protector = { s -> protect(s) })
                proxy = p; p.start()
            } catch (e: Exception) {
                Store.mitmEnabled = false
            }
        }

        Store.running = true
        reader = Thread({
            val buf = ByteArray(32767)
            try {
                while (!Thread.currentThread().isInterrupted) {
                    val n = inp.read(buf)
                    if (n > 0) eng.handle(buf, n)
                    else if (n < 0) break
                    else Thread.sleep(5)
                }
            } catch (e: Exception) {}
        }, "netlens-reader").apply { start() }
    }

    private fun stopVpn() {
        Store.running = false
        try { engine?.stop() } catch (e: Exception) {}
        engine = null
        try { proxy?.stop() } catch (e: Exception) {}
        proxy = null
        try { pfd?.close() } catch (e: Exception) {}
        pfd = null
        reader?.interrupt(); reader = null
        stopForeground(STOP_FOREGROUND_REMOVE)
        stopSelf()
    }

    override fun onRevoke() { stopVpn() }
    override fun onDestroy() { stopVpn(); super.onDestroy() }
}

        val all: List<PackageInfo> = try {
            if (android.os.Build.VERSION.SDK_INT >= 33) pm.getInstalledPackages(PackageManager.PackageInfoFlags.of(PackageManager.GET_PERMISSIONS.toLong()))
            else @Suppress("DEPRECATION") pm.getInstalledPackages(PackageManager.GET_PERMISSIONS)
        } catch (e: Exception) { emptyList() }
        for (pi in all) {
            val ai = pi.applicationInfo ?: continue
            val system = (ai.flags and ApplicationInfo.FLAG_SYSTEM) != 0 && (ai.flags and ApplicationInfo.FLAG_UPDATED_SYSTEM_APP) == 0
            if (system || pi.packageName == self) continue
            val req = pi.requestedPermissions ?: emptyArray()
            val fl = pi.requestedPermissionsFlags
            var internet = false
            val list = ArrayList<Pair<String, Boolean>>()
            for (i in req.indices) {
                val name = req[i]
                if (name == "android.permission.INTERNET") internet = true
                val human = SENSITIVE[name] ?: continue
                val g = fl != null && i < fl.size && (fl[i] and PackageInfo.REQUESTED_PERMISSION_GRANTED) != 0
                list.add(Pair(human, g))
            }
            val label = try { ai.loadLabel(pm).toString() } catch (e: Exception) { pi.packageName }
            res.add(Info(pi.packageName, label, internet, list))
        }
        res.sortWith(compareByDescending<Info> { it.granted }.thenBy { it.label.lowercase() })
        return res
    }

    fun forPackage(pm: PackageManager, pkg: String): Info? = scan(pm, "").firstOrNull { it.pkg == pkg }

package com.lab.netlens

import java.util.Collections
import java.util.concurrent.ConcurrentHashMap
import java.util.concurrent.atomic.AtomicInteger
import java.util.concurrent.atomic.AtomicLong

class AppStat(val pkg: String, val label: String) {
    val up = AtomicLong(); val down = AtomicLong()
    val flows = AtomicInteger(); val clearFlows = AtomicInteger()
    val mitmCount = AtomicInteger()
    val hosts: MutableSet<String> = Collections.newSetFromMap(ConcurrentHashMap<String, Boolean>())
    val trackers: MutableSet<String> = Collections.newSetFromMap(ConcurrentHashMap<String, Boolean>())
    val sensitive: MutableSet<String> = Collections.newSetFromMap(ConcurrentHashMap<String, Boolean>())
    val orgs: MutableSet<String> = Collections.newSetFromMap(ConcurrentHashMap<String, Boolean>())
}

class Flow(val id: Long, val stat: AppStat, val proto: String, val dstIp: String, val dstPort: Int) {
    val start: Long = System.currentTimeMillis()
    @Volatile var last: Long = start
    @Volatile var host: String? = null
    @Volatile var closed: Boolean = false
    @Volatile var mitm: Boolean = false
    @Volatile var org: String? = null
    val up = AtomicLong(); val down = AtomicLong()
    val notes: MutableSet<String> = Collections.synchronizedSet(LinkedHashSet<String>())
    fun addUp(n: Int) { up.addAndGet(n.toLong()); stat.up.addAndGet(n.toLong()); last = System.currentTimeMillis() }
    fun addDown(n: Int) { down.addAndGet(n.toLong()); stat.down.addAndGet(n.toLong()); last = System.currentTimeMillis() }
    fun noteHost(h: String) {
        if (h.isEmpty()) return
        host = h; stat.hosts.add(h)
        val c = Dpi.category(h)
        if (c != null) { notes.add("🎯 Posible destino de " + c.lowercase() + ": " + h); stat.trackers.add(c + "|" + h) }
    }
    fun name(): String = host ?: Store.dns[dstIp] ?: dstIp
}

object Store {
    private val lock = Any()
    private val list = ArrayList<Flow>()
    private val ids = AtomicLong()
    val apps = ConcurrentHashMap<String, AppStat>()
    val dns = ConcurrentHashMap<String, String>()
    val ipOrg = ConcurrentHashMap<String, String>()
    @Volatile var running = false
    @Volatile var blockQuic = false
    @Volatile var mitmEnabled = false

    fun app(pkg: String, label: String): AppStat = apps.getOrPut(pkg) { AppStat(pkg, label) }
    fun newFlow(stat: AppStat, proto: String, ip: String, port: Int): Flow {
        val f = Flow(ids.incrementAndGet(), stat, proto, ip, port)
        stat.flows.incrementAndGet()
        dns[ip]?.let { f.noteHost(it) }
        synchronized(lock) { list.add(f); while (list.size > 1200) list.removeAt(0) }
        return f
    }
    fun snapshot(): List<Flow> = synchronized(lock) { ArrayList(list) }
    fun clear() { synchronized(lock) { list.clear() }; apps.clear(); ipOrg.clear() }
}

# NetLens (laboratorio didáctico)

VPN local + relay TCP/UDP + DPI ligero + auditoría de permisos + **MITM TLS opcional (fase 2)**.
Todo en local, sin servidor.

## Fase 1
DNS, SNI, HTTP en claro, atribución por app, permisos.

## Fase 2 (MITM TLS)
1. Arranca NetLens una vez para que se genere la CA local.
2. Pulsa **Exportar CA**. El certificado `netlens-ca.crt` se guarda en Descargas.
3. Instálalo: Ajustes → Seguridad → Cifrado y credenciales → Instalar certificado → **Certificado de CA** → selecciona el archivo.
4. Comprueba la huella SHA-256 que muestra NetLens tras exportar.
5. Activa **MITM TLS** en NetLens.
6. Solo funcionará con apps que acepten CAs de usuario (por ejemplo, tus propias apps compiladas en debug con `network_security_config.xml` que incluya `<certificates src="user"/>` dentro de `<debug-overrides>`).
7. Las apps con certificate pinning rechazarán el certificado: comportamiento correcto. No se salta pinning.

## Alcance legal
Solo tu dispositivo, tus apps, tu tráfico. Interceptar tráfico de terceros o de redes ajenas es ilegal.

## Limitaciones conocidas
- Solo se hace MITM de TCP:443. Otros puertos van directos.
- QUIC (UDP:443) no se descifra. Usa «Bloquear QUIC» para forzar TCP+TLS.
- TLS 1.3 con ECH no se puede descifrar (el SNI va cifrado).
- Apps con certificate pinning fallarán el handshake.

package com.lab.netlens

import org.json.JSONObject
import java.net.HttpURLConnection
import java.net.URL
import java.util.concurrent.ConcurrentHashMap

object IpInfo {
    private val cache = ConcurrentHashMap<String, String>()

    fun org(ip: String): String? = cache[ip]

    fun resolve(ips: Collection<String>, onDone: () -> Unit) {
        Thread {
            for (ip in ips) {
                if (cache.containsKey(ip)) continue
                try {
                    val url = URL("https://ip-api.com/json/$ip?fields=status,country,isp,as,org")
                    val conn = url.openConnection() as HttpURLConnection
                    conn.connectTimeout = 5000
                    conn.readTimeout = 5000
                    conn.setRequestProperty("User-Agent", "NetLens/1.0")
                    val text = conn.inputStream.bufferedReader().use { it.readText() }
                    val json = JSONObject(text)
                    if (json.optString("status") == "success") {
                        val org = json.optString("org", "").ifEmpty { json.optString("isp", "") }
                        val asn = json.optString("as", "")
                        val country = json.optString("country", "")
                        val value = listOf(org, asn, country).filter { it.isNotEmpty() }.joinToString(" · ")
                        if (value.isNotEmpty()) {
                            cache[ip] = value
                            for (f in Store.snapshot()) {
                                if (f.dstIp == ip) {
                                    f.org = value
                                    f.stat.orgs.add(value)
                                }
                            }
                        }
                    }
                } catch (_: Exception) {
                }
            }
            onDone()
        }.start()
    }
}

package com.lab.netlens

import android.content.Context
import android.content.SharedPreferences
import java.util.Locale
import java.util.concurrent.ConcurrentHashMap
import java.util.concurrent.atomic.AtomicLong

object Protection {
    private lateinit var prefs: SharedPreferences
    @Volatile var rules: Set<String> = emptySet()
        private set
    val blocked = AtomicLong()
    val byApp = ConcurrentHashMap<String, AtomicLong>()
    var enabled: Boolean
        get() = prefs.getBoolean("enabled", false)
        set(value) { prefs.edit().putBoolean("enabled", value).apply() }
    fun init(ctx: Context) {
        prefs = ctx.applicationContext.getSharedPreferences("protection", Context.MODE_PRIVATE)
        rules = prefs.getStringSet("rules", emptySet())?.toSet() ?: emptySet()
    }
    fun save(text: String) {
        val next = text.lines().map { it.trim().lowercase(Locale.ROOT) }
            .filter { it.isNotEmpty() && !it.startsWith("#") }.toSet()
        require(next.all { Regex("[+-]([^| ]+\\|)?[a-z0-9.-]+").matches(it) }) {
            "Formato: -dominio.com, +dominio.com o -paquete|dominio.com"
        }
        prefs.edit().putStringSet("rules", next).apply(); rules = next
    }
    fun matches(host: String, rule: String): Boolean = host == rule || host.endsWith("." + rule)
    fun reason(pkg: String, host: String?): String? {
        if (!enabled || host.isNullOrBlank()) return null
        val h = host.lowercase(Locale.ROOT).trimEnd('.')
        val applicable = rules.filter { val r=it.drop(1); !r.contains('|') || r.substringBefore('|') == pkg }
        fun domain(r: String) = r.drop(1).substringAfter('|')
        if (applicable.any { it.startsWith("+") && matches(h, domain(it)) }) return null
        if (applicable.any { it.startsWith("-") && matches(h, domain(it)) }) return "Regla personalizada: $h"
        val category = Dpi.category(h)
        return if (category != null && !category.contains("fallos", true)) "Lista integrada: $category / $h" else null
    }
    fun record(pkg: String) { blocked.incrementAndGet(); byApp.getOrPut(pkg) { AtomicLong() }.incrementAndGet() }
    fun stats(): String = "Decisiones de bloqueo en este proceso: ${blocked.get()}\n" +
        byApp.entries.sortedByDescending { it.value.get() }.joinToString("\n") { "${it.key}: ${it.value.get()}" } +
        "\nNo son bytes ahorrados ni fugas confirmadas. Reiniciar el proceso reinicia contadores."
}
